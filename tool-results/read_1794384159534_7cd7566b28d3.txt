     1→// Market scanner engine.
     2→// A singleton 5s loop that:
     3→//  - walks simulated prices for the 6 indices + NSE F&O stock universe (paper)
     4→//  - pulls real quotes via Shoonya when a LIVE scanner is enabled
     5→//  - builds rolling candles and evaluates spot price-action patterns:
     6→//      LIQUIDITY_SWEEP · ORDER_BLOCK · HEAD_SHOULDERS · DOUBLE_BOTTOM · DOUBLE_TOP
     7→//    plus tick-level ORB and MOMENTUM
     8→//  - every signal carries explicit entry / SL / target levels on the spot price
     9→//  - optionally auto-places: equity market orders for stocks, ATM option
    10→//    (CE on LONG / PE on SHORT) for indices, then manages SL / target /
    11→//    timed square-off and premium P&L for index trades
    12→

    97→function round2(x: number): number {
    98→  return Math.round(x * 100) / 100;

   101→/** Parse a scanner-executed option tradingsymbol back into its parts. */
   102→function parseExecSymbol(tsym: string):
   103→  | { instId: InstrumentId; strike: number; type: "CE" | "PE"; settleMs: number }
   104→  | null {
   105→  const m = tsym.match(/^(NIFTY|BANKNIFTY|FINNIFTY|MIDCPNIFTY|SENSEX|BANKEX)(\d{2})([A-Z]{3})(\d{2})(\d+)(CE|PE)$/);
   106→  if (!m) return null;
   107→  const months = ["JAN", "FEB", "MAR", "APR", "MAY", "JUN", "JUL", "AUG", "SEP", "OCT", "NOV", "DEC"];
   108→  const mi = months.indexOf(m[3]);
   109→  if (mi < 0) return null;
   110→  const settleMs = Date.UTC(2000 + parseInt(m[4], 10), mi, parseInt(m[2], 10), 10, 0, 0);
   111→  return {
   112→    instId: m[1] as InstrumentId,
   113→    strike: parseFloat(m[5]),
   114→    type: m[6] as "CE" | "PE",
   115→    settleMs,
   116→  };
   117→}

   136→  private timer: ReturnType<typeof setInterval> | null = null;
   137→
   138→  start() {
   139→    if (this.timer) return;
   140→    for (const a of SCAN_ASSETS) this.state.prices.set(a.symbol, a.basePrice);
   141→    console.log(`[scanner] started v${this.codeVersion} (${SCAN_ASSETS.length} symbols incl. indices)`);
   142→    this.timer = setInterval(() => {
   143→      this.tick().catch((e) => console.error("[scanner] tick error:", e));
   144→    }, SCAN_TICK_MS);
   145→  }

   147→  stop() {
   148→    if (this.timer) {
   149→      clearInterval(this.timer);
   150→      this.timer = null;
   151→    }
   152→  }
   153→
   154→  price(symbol: string): number {
   155→    return this.state.prices.get(symbol) ?? SCAN_ASSET_MAP[symbol]?.basePrice ?? 0;
   156→  }

   158→  private rollDay(day: string) {
   159→    if (this.state.day !== day) {
   160→      this.state.day = day;
   161→      this.state.dayOpen.clear();
   162→      this.state.orbHigh.clear();
   163→      this.state.orbLow.clear();
   164→      this.state.orbTicks.clear();
   165→      this.state.cooldown.clear();
   166→      this.state.candles.clear();
   167→      this.state.cur.clear();
   168→      this.state.backfilled.clear();
   169→    }
   170→  }
   171→
   172→  private walk() {
   173→    for (const a of SCAN_ASSETS) {
   174→      const p = this.state.prices.get(a.symbol) ?? a.basePrice;
   175→      // ~0.08% of price per 5s tick scaled by each symbol's vol -> candles move
   176→      const next = p + (a.basePrice - p) * 0.0004 + p * a.vol * 0.0035 * gaussian();
   177→      const snapped = round2(next);
   178→      this.state.prices.set(a.symbol, snapped);
   179→      if (!this.state.dayOpen.has(a.symbol)) this.state.dayOpen.set(a.symbol, snapped);
   180→    }
   181→  }

   185→  /** Seed paper candle history (bridged random walk ending at the current price). */
   186→  private backfill(sym: string, asset: ScanAsset, candleSec: number) {
   187→    const key = `${sym}:${candleSec}`;
   188→    if (this.state.backfilled.has(key)) return;
   189→    this.state.backfilled.add(key);
   190→    const price = this.price(sym) || asset.basePrice;
   191→    const sigma = (price * asset.vol) / Math.sqrt(252 * 375) * Math.sqrt(candleSec / 60);
   192→    const steps: number[] = [0];
   193→    let cum = 0;
   194→    for (let i = 1; i <= BACKFILL_BARS; i++) {
   195→      cum += gaussian();
   196→      steps.push(cum);
   197→    }
   198→    const rawN = steps[BACKFILL_BARS];
   199→    const candles: Candle[] = [];
   200→    const nowMs = Date.now();
   201→    const grid = Math.floor(nowMs / (candleSec * 1000)) * candleSec * 1000;
   202→    let prev = price; // bridged series ends exactly at the current price
   203→    for (let i = 1; i <= BACKFILL_BARS; i++) {
   204→      const c = price + (steps[i] - (i / BACKFILL_BARS) * rawN) * sigma;
   205→      const o = prev;
   206→      const wickUp = Math.abs(gaussian()) * sigma * 0.5;
   207→      const wickDn = Math.abs(gaussian()) * sigma * 0.5;
   208→      candles.push({
   209→        t: grid - (BACKFILL_BARS - i) * candleSec * 1000,
   210→        o: round2(o),
   211→        h: round2(Math.max(o, c) + wickUp),
   212→        l: round2(Math.min(o, c) - wickDn),
   213→        c: round2(c),
   214→      });
   215→      prev = c;
   216→    }
   217→    this.state.candles.set(sym, candles);
   218→  }
   219→
   220→  /** Append a price tick to the in-progress candle; returns true when a candle just closed. */
   221→  private appendTick(sym: string, price: number, candleSec: number): boolean {
   222→    const nowMs = Date.now();
   223→    const start = Math.floor(nowMs / (candleSec * 1000)) * candleSec * 1000;
   224→    const cur = this.state.cur.get(sym);
   225→    if (!cur || cur.t !== start) {
   226→      if (cur && cur.n > 0) {
   227→        const arr = this.state.candles.get(sym) ?? [];
   228→        arr.push({ t: cur.t, o: cur.o, h: cur.h, l: cur.l, c: cur.c });
   229→        if (arr.length > MAX_CANDLES) arr.splice(0, arr.length - MAX_CANDLES);
   230→        this.state.candles.set(sym, arr);
   231→        this.state.cur.delete(sym);
   232→        // start the new candle
   233→        this.state.cur.set(sym, { t: start, o: price, h: price, l: price, c: price, n: 1 });
   234→        return true;
   235→      }
   236→      this.state.cur.set(sym, { t: start, o: price, h: price, l: price, c: price, n: 1 });
   237→      return false;
   238→    }
   239→    cur.h = Math.max(cur.h, price);
   240→    cur.l = Math.min(cur.l, price);
   241→    cur.c = price;
   242→    cur.n++;
   243→    return false;
   244→  }

   246→  // ---------- live quotes ----------
   247→  private async ensureSession(): Promise<boolean> {
   248→    if (this.state.session) return true;
   249→    const acct = await db.brokerAccount.findUnique({ where: { id: "shoonya" } });
   250→    if (!acct || !acct.userId || !acct.password || !acct.apiKey) return false;
   251→    const creds: ShoonyaCreds = {
   252→      userId: acct.userId,
   253→      password: acct.password,
   254→      factor2: acct.factor2 ?? "",
   255→      vendorCode: acct.vendorCode ?? "",
   256→      apiKey: acct.apiKey,
   257→      imei: acct.imei ?? "shoonyaalgo",
   258→      totpSecret: acct.totpSecret,
   259→    };
   260→    const res = await shoonyaLogin(creds);
   261→    if (res.ok && res.token) {
   262→      this.state.session = { token: res.token, actid: res.account ?? acct.userId, userId: acct.userId };
   263→      return true;
   264→    }
   265→    return false;
   266→  }

   268→  private throttleError(message: string) {
   269→    const now = Date.now();
   270→    if (now - this.state.lastErrorAt < ERROR_THROTTLE_MS) return;
   271→    this.state.lastErrorAt = now;
   272→    void db.event
   273→      .create({ data: { deploymentId: null, type: "ERROR", message, level: "ERROR" } })
   274→      .catch(() => {});
   275→  }

   279→  /** ATM option chosen for a directional index trade (CE on LONG, PE on SHORT). */
   280→  private indexExecution(
   281→    instId: InstrumentId,
   282→    spot: number,
   283→    side: "LONG" | "SHORT",
   284→    now: Date,
   285→  ): { execSymbol: string; exchange: "NFO" | "BFO"; execQty: number; strike: number; type: "CE" | "PE"; settleMs: number; lotSize: number; iv: number } | null {
   286→    const inst = INSTRUMENTS[instId];
   287→    if (!inst) return null;
   288→    const expiry = resolveExpiry(inst, "NEAREST", now).date;
   289→    const settleMs = Date.UTC(expiry.getUTCFullYear(), expiry.getUTCMonth(), expiry.getUTCDate(), 10, 0, 0);
   290→    const strike = roundStrike(spot, inst.strikeStep);
   291→    const type = side === "LONG" ? "CE" : "PE";
   292→    return {
   293→      execSymbol: optionSymbol(inst, expiry, strike, type),
   294→      exchange: inst.exchange,
   295→      execQty: inst.lotSize, // 1 lot; sized up below if capital allows
   296→      strike,
   297→      type,
   298→      settleMs,
   299→      lotSize: inst.lotSize,
   300→      iv: inst.iv,
   301→    };
   302→  }
   303→
   304→  private bsPremiumFor(exec: NonNullable<ReturnType<Scanner["indexExecution"]>>, spot: number): number {
   305→    const T = Math.max((exec.settleMs - Date.now()) / MS_PER_YEAR, 0);
   306→    return roundTick(paperPremium({ spot, strike: exec.strike, iv: exec.iv, yearsToExpiry: T, type: exec.type }));
   307→  }

   309→  // ---------- main tick ----------
   310→  async tick(): Promise<void> {
   311→    if (this.state.ticking) return;
   312→    this.state.ticking = true;
   313→    try {
   314→      const now = istNow();
   315→      this.rollDay(`${now.yyyy}-${now.mm}-${now.dd}`);
   316→      const marketOpen = isMarketOpen(now);
   317→
   318→      const config = await db.scannerConfig.findUnique({ where: { id: "global" } });
   319→      if (!config) return;
   320→
   321→      const enabledSymbols = config.symbols
   322→        .split(",")
   323→        .map((s) => s.trim().toUpperCase())
   324→        .filter((s) => SCAN_ASSET_MAP[s] !== undefined);
   325→
   326→      const live = config.mode === "LIVE";
   327→      const candleSec = Math.max(15, Math.min(300, config.candleSeconds || 60));
   328→
   329→      // ---------- price source ----------
   330→      if (live && enabledSymbols.length > 0) {
   331→        const ok = await this.ensureSession();
   332→        if (!ok) {
   333→          this.throttleError("Scanner LIVE mode: no broker session — quotes unavailable. Configure Shoonya credentials in Settings.");
   334→          return;
   335→        }
   336→        const results = await Promise.all(
   337→          enabledSymbols.map(async (sym) => {
   338→            const asset = SCAN_ASSET_MAP[sym];
   339→            const lp =
   340→              asset?.kind === "INDEX"
   341→                ? await shoonyaQuote(this.state.session!.token, this.state.session!.userId, asset.exchange ?? "NFO", sym)
   342→                : await shoonyaQuote(this.state.session!.token, this.state.session!.userId, "NSE", `${sym}-EQ`);
   343→            return [sym, lp] as const;
   344→          }),
   345→        );
   346→        let anyQuote = false;
   347→        for (const [sym, lp] of results) {
   348→          if (lp !== null && lp > 0) {
   349→            this.state.prices.set(sym, lp);
   350→            if (!this.state.dayOpen.has(sym)) this.state.dayOpen.set(sym, lp);
   351→            anyQuote = true;
   352→          }
   353→        }
   354→        if (!anyQuote) {
   355→          this.throttleError("Scanner LIVE mode: quotes unreachable from this machine. Check network / broker session.");
   356→          return;
   357→        }
   358→      } else {
   359→        // paper (and pre-warm of sim sessions): walk simulated prices
   360→        this.walk();
   361→      }
   362→
   363→      if (!config.enabled || enabledSymbols.length === 0) return;
   364→
   365→      // ---------- manage open scan trades ----------
   366→      await this.manageOpenTrades(config, now, marketOpen);
   367→
   368→      // ---------- candle building ----------
   369→      const freshlyClosed = new Set<string>();
   370→      for (const sym of enabledSymbols) {
   371→        const asset = SCAN_ASSET_MAP[sym];
   372→        if (!asset) continue;
   373→        const price = this.price(sym);
   374→        if (!price) continue;
   375→        if (!live) this.backfill(sym, asset, candleSec);
   376→        if (this.appendTick(sym, price, candleSec)) freshlyClosed.add(sym);
   377→      }
   378→
   379→      // ---------- evaluate signals ----------
   380→      const openTrades = await db.scanTrade.findMany({ where: { status: "OPEN" } });
   381→      const inTrade = new Set(openTrades.map((t) => t.symbol));
   382→
   383→      let strategies: string[] = [];
   384→      try {
   385→        const parsed = JSON.parse(config.strategies) as unknown;
   386→        if (Array.isArray(parsed)) strategies = parsed.filter((s) => validStrategyIds().includes(String(s)));
   387→      } catch {
   388→        strategies = ["ORB"];
   389→      }
   390→      if (config.strategy === "MOMENTUM" && strategies.length === 1 && strategies[0] === "ORB") strategies = ["MOMENTUM"];
   391→      const patternIds = strategies.filter((s) => !["ORB", "MOMENTUM"].includes(s)) as PatternId[];
   392→      let overrides: Record<string, Record<string, number>> = {};
   393→      try {
   394→        overrides = JSON.parse(config.patternParams) as typeof overrides;
   395→      } catch {
   396→        overrides = {};
   397→      }
   398→
   399→      for (const sym of enabledSymbols) {
   400→        const asset = getScanAsset(sym);
   401→        if (!asset) continue;
   402→        const price = this.price(sym);
   403→        const open = this.state.dayOpen.get(sym);
   404→        if (!open) continue;
   405→
   406→        // opening-range bookkeeping (ORB / momentum use tick ranges)
   407→        const t = (this.state.orbTicks.get(sym) ?? 0) + 1;
   408→        this.state.orbTicks.set(sym, t);
   409→        if (t <= ORB_TICKS) {
   410→          this.state.orbHigh.set(sym, Math.max(this.state.orbHigh.get(sym) ?? price, price));
   411→          this.state.orbLow.set(sym, Math.min(this.state.orbLow.get(sym) ?? price, price));
   412→        }
   413→
   414→        if (this.state.cooldown.has(sym) || inTrade.has(sym)) continue;
   415→
   416→        // ----- pattern signals (candle based) -----
   417→        if (patternIds.length > 0 && freshlyClosed.has(sym)) {
   418→          const candles = this.state.candles.get(sym) ?? [];
   419→          const sigs = runPatternDetectors(candles, patternIds, overrides);
   420→          if (sigs.length > 0) {
   421→            const s = sigs[0];
   422→            await this.fireSignal(config, asset, sym, {
   423→              strategy: s.strategy,
   424→              side: s.side,
   425→              entry: round2(s.entry),
   426→              sl: round2(s.sl),
   427→              target: round2(s.target),
   428→              ref: round2(s.ref),
   429→              note: s.note,
   430→            });
   431→            continue;
   432→          }
   433→        }
   434→
   435→        // ----- base strategies (tick based) -----
   436→        let spec: ScanTradeSpec | null = null;
   437→        if (strategies.includes("ORB")) {
   438→          if (t > ORB_TICKS) {
   439→            const hi = this.state.orbHigh.get(sym);
   440→            const lo = this.state.orbLow.get(sym);
   441→            if (hi && lo) {
   442→              const b = round2(price * 0.0005);
   443→              if (price > hi + b) {
   444→                spec = {
   445→                  strategy: "ORB",
   446→                  side: "LONG",
   447→                  entry: round2(price),
   448→                  sl: round2(price * (1 - config.slPct / 100)),
   449→                  target: round2(price * (1 + config.targetPct / 100)),
   450→                  ref: round2(hi),
   451→                  note: `Broke above opening range high ₹${hi.toFixed(2)} (range ₹${lo.toFixed(2)}–₹${hi.toFixed(2)}).`,
   452→                };
   453→              } else if (price < lo - b) {
   454→                spec = {
   455→                  strategy: "ORB",
   456→                  side: "SHORT",
   457→                  entry: round2(price),
   458→                  sl: round2(price * (1 + config.slPct / 100)),
   459→                  target: round2(price * (1 - config.targetPct / 100)),
   460→                  ref: round2(lo),
   461→                  note: `Broke below opening range low ₹${lo.toFixed(2)} (range ₹${lo.toFixed(2)}–₹${hi.toFixed(2)}).`,
   462→                };
   463→              }
   464→            }
   465→          }
   466→        } else if (strategies.includes("MOMENTUM") && spec === null) {
   467→          if (t > MOMENTUM_WARMUP_TICKS) {
   468→            const chg = price / open - 1;
   469→            if (chg >= MOMENTUM_THRESHOLD) {
   470→              spec = {
   471→                strategy: "MOMENTUM",
   472→                side: "LONG",
   473→                entry: round2(price),
   474→                sl: round2(price * (1 - config.slPct / 100)),
   475→                target: round2(price * (1 + config.targetPct / 100)),
   476→                ref: round2(open),
   477→                note: `Up ${((price / open - 1) * 100).toFixed(2)}% from day open — momentum continuation.`,
   478→              };
   479→            } else if (chg <= -MOMENTUM_THRESHOLD) {
   480→              spec = {
   481→                strategy: "MOMENTUM",
   482→                side: "SHORT",
   483→                entry: round2(price),
   484→                sl: round2(price * (1 + config.slPct / 100)),
   485→                target: round2(price * (1 - config.targetPct / 100)),
   486→                ref: round2(open),
   487→                note: `Down ${(Math.abs(chg) * 100).toFixed(2)}% from day open — momentum breakdown.`,
   488→              };
   489→            }
   490→          }
   491→        }
   492→        if (spec) await this.fireSignal(config, asset, sym, spec);
   493→      }
   494→    } finally {
   495→      this.state.ticking = false;
   496→    }
   497→  }

   499→  /** Record a signal and (optionally) place the trade with explicit spot levels. */
   500→  private async fireSignal(
   501→    config: {
   502→      autoPlace: boolean;
   503→      mode: string;
   504→      capitalPerTrade: number;
   505→    },
   506→    asset: ScanAsset,
   507→    sym: string,
   508→    spec: ScanTradeSpec,
   509→  ) {
   510→    this.state.cooldown.add(sym);
   511→    let placed = false;
   512→    let tradeId: string | null = null;
   513→
   514→    if (config.autoPlace) placed = true; // set false again if execution fails outright
   515→
   516→    let tradeRow: { id: string } | null = null;
   517→    if (config.autoPlace) {
   518→      try {
   519→        tradeRow = await this.openScanTrade(config, asset, sym, spec);
   520→        tradeId = tradeRow?.id ?? null;
   521→        placed = tradeRow !== null;
   522→      } catch (e) {
   523→        console.error("[scanner] trade placement failed:", e);
   524→        placed = false;
   525→      }
   526→    }
   527→
   528→    await db.scanSignal.create({
   529→      data: {
   530→        symbol: sym,
   531→        strategy: spec.strategy,
   532→        side: spec.side,
   533→        refPrice: spec.ref,
   534→        price: spec.entry,
   535→        entryLevel: spec.entry,
   536→        slLevel: spec.sl,
   537→        targetLevel: spec.target,
   538→        note: spec.note,
   539→        placed,
   540→        tradeId,
   541→      },
   542→    });
   543→  }

   545→  /** Create the ScanTrade (equity cash or index ATM option) for a signal. */
   546→  private async openScanTrade(
   547→    config: { mode: string; capitalPerTrade: number },
   548→    asset: ScanAsset,
   549→    sym: string,
   550→    spec: ScanTradeSpec,
   551→  ): Promise<{ id: string } | null> {
   552→    const now = new Date();
   553→    let qty = 0;
   554→    let execSymbol: string | null = null;
   555→    let execQty = 0;
   556→    let entryPremium: number | null = null;
   557→    let orderTsyl = `${sym}-EQ`;
   558→    let orderExchange = "NSE";
   559→    let orderPrice = spec.entry;
   560→    let execDesc = `${sym} × qty`;
   561→
   562→    if (asset.kind === "INDEX") {
   563→      const instId = sym as InstrumentId;
   564→      const exec = this.indexExecution(instId, spec.entry, spec.side, now);
   565→      if (!exec) return null;
   566→      const est = this.bsPremiumFor(exec, spec.entry);
   567→      // live: prefer the real option LTP when available
   568→      let premium = est;
   569→      if (config.mode === "LIVE" && this.state.session) {
   570→        const lp = await shoonyaQuote(this.state.session.token, this.state.session.userId, exec.exchange, exec.execSymbol).catch(() => null);
   571→        if (lp && lp > 0) premium = lp;
   572→      }
   573→      const lots = Math.max(1, Math.floor(config.capitalPerTrade / Math.max(1, premium * exec.lotSize)));
   574→      execQty = lots * exec.lotSize;
   575→      execSymbol = exec.execSymbol;
   576→      entryPremium = premium;
   577→      qty = 0;
   578→      orderTsyl = exec.execSymbol;
   579→      orderExchange = exec.exchange;
   580→      orderPrice = premium;
   581→      execDesc = `${exec.execSymbol} × ${execQty} (${lots} lot${lots > 1 ? "s" : ""}) @ ₹${premium.toFixed(2)}`;
   582→    } else {
   583→      qty = Math.max(1, Math.floor(config.capitalPerTrade / spec.entry));
   584→      execDesc = `${sym} × ${qty}`;
   585→    }
   586→
   587→    const entry = asset.kind === "EQUITY" ? roundTick(spec.entry * (1 + (spec.side === "LONG" ? SLIPPAGE : -SLIPPAGE))) : round2(spec.entry);
   588→
   589→    const trade = await db.scanTrade.create({
   590→      data: {
   591→        symbol: sym,
   592→        kind: asset.kind,
   593→        strategy: spec.strategy,
   594→        side: spec.side,
   595→        qty,
   596→        entryPrice: entry,
   597→        ltp: entry,
   598→        slPrice: spec.sl,
   599→        targetPrice: spec.target,
   600→        execSymbol,
   601→        execQty,
   602→        entryPremium,
   603→        ltpPremium: entryPremium,
   604→        mode: config.mode,
   605→        status: "OPEN",
   606→      },
   607→    });
   608→
   609→    await db.order.create({
   610→      data: {
   611→        deploymentId: null,
   612→        tsym: orderTsyl,
   613→        exchange: orderExchange,
   614→        side: spec.side === "LONG" ? "B" : "S",
   615→        qty: asset.kind === "EQUITY" ? qty : execQty,
   616→        price: orderPrice,
   617→        priceType: "MKT",
   618→        mode: config.mode,
   619→        status: "FILLED",
   620→        message:
   621→          config.mode === "LIVE"
   622→            ? `Live scanner order placed (${execDesc})`
   623→            : `Scanner paper fill @ ${asset.kind === "INDEX" ? `₹${(entryPremium ?? 0).toFixed(2)} prem` : `₹${entry.toFixed(2)}`}`,
   624→      },
   625→    });
   626→    await db.event
   627→      .create({
   628→        data: {
   629→          deploymentId: null,
   630→          type: "SCAN_ENTRY",
   631→          level: "TRADE",
   632→          message: `Scanner ${config.mode} ${spec.side} ${sym} (${spec.strategy}) entry ₹${spec.entry.toFixed(2)} — SL ₹${spec.sl.toFixed(2)}, target ₹${spec.target.toFixed(2)}. Exec: ${execDesc}.`,
   633→        },
   634→      })
   635→      .catch(() => {});
   636→
   637→    if (config.mode === "LIVE" && this.state.session) {
   638→      // fire the real order (best effort; fills tracked via broker)
   639→      await shoonyaPlaceOrder(
   640→        this.state.session.token,
   641→        this.state.session.actid,
   642→        this.state.session.userId,
   643→        {
   644→          exchange: orderExchange,
   645→          symbol: orderTsyl,
   646→          qty: asset.kind === "EQUITY" ? qty : execQty,
   647→          side: spec.side === "LONG" ? "B" : "S",
   648→          price: 0,
   649→          priceType: "MKT",
   650→          product: asset.kind === "EQUITY" ? "I" : "M",
   651→          remarks: "SA-SCAN",
   652→        },
   653→      ).catch(() => null);
   654→    }
   655→
   656→    return { id: trade.id };
   657→  }

   661→  private async manageOpenTrades(
   662→    config: { slPct: number; targetPct: number; squareOffTime: string; mode: string },
   663→    now: ReturnType<typeof istNow>,
   664→    marketOpen: boolean,
   665→  ) {
   666→    const open = await db.scanTrade.findMany({ where: { status: "OPEN" } });
   667→    const nowMin = now.hh * 60 + now.minute;
   668→    const soMin = hhmmToMinutes(config.squareOffTime);
   669→
   670→    for (const trade of open) {
   671→      const spot = this.price(trade.symbol);
   672→      if (!spot) continue;
   673→      const isIndex = trade.kind === "INDEX";
   674→
   675→      let closeStatus: string | null = null;
   676→      let closeMsg = "";
   677→      if (
   678→        trade.slPrice &&
   679→        ((trade.side === "LONG" && spot <= trade.slPrice) || (trade.side === "SHORT" && spot >= trade.slPrice))
   680→      ) {
   681→        closeStatus = "CLOSED_SL";
   682→        closeMsg = `Stop-loss level ₹${trade.slPrice.toFixed(2)} hit`;
   683→      } else if (
   684→        trade.targetPrice &&
   685→        ((trade.side === "LONG" && spot >= trade.targetPrice) || (trade.side === "SHORT" && spot <= trade.targetPrice))
   686→      ) {
   687→        closeStatus = "CLOSED_TARGET";
   688→        closeMsg = `Target level ₹${trade.targetPrice.toFixed(2)} reached`;
   689→      } else if (marketOpen && nowMin >= soMin) {
   690→        closeStatus = "CLOSED_TIME";
   691→        closeMsg = `Timed square-off at ${config.squareOffTime} IST`;
   692→      }
   693→
   694→      if (closeStatus) {
   695→        const exit = isIndex ? round2(spot) : roundTick(spot * (1 + (trade.side === "LONG" ? -SLIPPAGE : SLIPPAGE)));
   696→        let exitPremium: number | null = null;
   697→        let pnl = 0;
   698→        if (isIndex && trade.execSymbol && trade.entryPremium) {
   699→          const parsed = parseExecSymbol(trade.execSymbol);
   700→          if (parsed) {
   701→            const inst = INSTRUMENTS[parsed.instId];
   702→            const exec = {
   703→              execSymbol: trade.execSymbol,
   704→              exchange: inst.exchange,
   705→              execQty: trade.execQty,
   706→              strike: parsed.strike,
   707→              type: parsed.type,
   708→              settleMs: parsed.settleMs,
   709→              lotSize: inst.lotSize,
   710→              iv: inst.iv,
   711→            };
   712→            exitPremium = this.bsPremiumFor(exec, spot);
   713→            if (config.mode === "LIVE" && this.state.session) {
   714→              const lp = await shoonyaQuote(this.state.session.token, this.state.session.userId, inst.exchange, trade.execSymbol).catch(() => null);
   715→              if (lp && lp > 0) exitPremium = lp;
   716→            }
   717→            pnl = Math.round((exitPremium - trade.entryPremium) * trade.execQty * 100) / 100;
   718→          }
   719→        } else {
   720→          pnl = Math.round((trade.side === "LONG" ? 1 : -1) * (exit - trade.entryPrice) * trade.qty * 100) / 100;
   721→        }
   722→        await db.scanTrade.update({
   723→          where: { id: trade.id },
   724→          data: {
   725→            status: closeStatus,
   726→            exitPrice: exit,
   727→            ltp: exit,
   728→            exitPremium,
   729→            ltpPremium: exitPremium,
   730→            realizedPnl: pnl,
   731→            closedAt: new Date(),
   732→          },
   733→        });
   734→        await db.order.create({
   735→          data: {
   736→            deploymentId: null,
   737→            tsym: isIndex && trade.execSymbol ? trade.execSymbol : `${trade.symbol}-EQ`,
   738→            exchange: isIndex ? INSTRUMENTS[trade.symbol as InstrumentId]?.exchange ?? "NFO" : "NSE",
   739→            side: trade.side === "LONG" ? "S" : "B",
   740→            qty: isIndex ? trade.execQty : trade.qty,
   741→            price: isIndex ? exitPremium ?? 0 : exit,
   742→            priceType: "MKT",
   743→            mode: trade.mode,
   744→            status: "FILLED",
   745→            message: `Scanner exit (${closeStatus}) @ ${isIndex ? `₹${(exitPremium ?? 0).toFixed(2)} prem` : `₹${exit.toFixed(2)}`}`,
   746→          },
   747→        });
   748→        await db.event
   749→          .create({
   750→            data: {
   751→              deploymentId: null,
   752→              type: "SCAN_EXIT",
   753→              level: "TRADE",
   754→              message: `Scanner ${trade.side} ${trade.symbol} closed — ${closeMsg}. P&L ₹${pnl.toLocaleString("en-IN")}.`,
   755→            },
   756→          })
   757→          .catch(() => {});
   758→      } else {
   759→        const upd: Record<string, number> = {};
   760→        if (trade.ltp !== spot) upd.ltp = spot;
   761→        if (isIndex && trade.execSymbol) {
   762→          const parsed = parseExecSymbol(trade.execSymbol);
   763→          if (parsed) {
   764→            const inst = INSTRUMENTS[parsed.instId];
   765→            const prem = this.bsPremiumFor(
   766→              {
   767→                execSymbol: trade.execSymbol,
   768→                exchange: inst.exchange,
   769→                execQty: trade.execQty,
   770→                strike: parsed.strike,
   771→                type: parsed.type,
   772→                settleMs: parsed.settleMs,
   773→                lotSize: inst.lotSize,
   774→                iv: inst.iv,
   775→              },
   776→              spot,
   777→            );
   778→            if (prem !== trade.ltpPremium) upd.ltpPremium = prem;
   779→          }
   780→        }
   781→        if (Object.keys(upd).length > 0) {
   782→          await db.scanTrade.update({ where: { id: trade.id }, data: upd });
   783→        }
   784→      }
   785→    }
   786→    void config;
   787→  }

   870→  universe(enabledSet: Set<string>, inTradeSet: Set<string>): ScannerUniverseRow[] {
   871→    const rows: ScannerUniverseRow[] = [];
   872→    for (const a of SCAN_ASSETS) {
   873→      const price = this.price(a.symbol);
   874→      const open = this.state.dayOpen.get(a.symbol) ?? price;
   875→      const hi = this.state.orbHigh.get(a.symbol) ?? null;
   876→      const lo = this.state.orbLow.get(a.symbol) ?? null;
   877→      const ticks = this.state.orbTicks.get(a.symbol) ?? 0;
   878→      let state: ScannerUniverseRow["state"] = "IDLE";
   879→      if (inTradeSet.has(a.symbol)) state = "IN_TRADE";
   880→      else if (this.state.cooldown.has(a.symbol)) state = "COOLDOWN";
   881→      else if (enabledSet.has(a.symbol)) state = "SCANNING";
   882→      rows.push({
   883→        symbol: a.symbol,
   884→        name: a.name,
   885→        kind: a.kind,
   886→        price: Math.round(price * 100) / 100,
   887→        dayOpen: Math.round(open * 100) / 100,
   888→        chgPct: open ? Math.round((price / open - 1) * 10000) / 100 : 0,
   889→        orbHigh: hi,
   890→        orbLow: lo,
   891→        rangeReady: ticks > ORB_TICKS,
   892→        bars: (this.state.candles.get(a.symbol) ?? []).length,
   893→        enabled: enabledSet.has(a.symbol),
   894→        state,
   895→      });
   896→    }
   897→    return rows;
   898→  }
